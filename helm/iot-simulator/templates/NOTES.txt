Check pod status:

  kubectl get pods -n {{ .Release.Namespace }} \

View logs:

  kubectl logs -n {{ .Release.Namespace }} \
    -l "app.kubernetes.io/instance={{ .Release.Name }}" --follow

Upgrade:

  helm upgrade {{ .Release.Name }} ./helm/iot-simulator -n {{ .Release.Namespace }}

Uninstall:

  helm uninstall {{ .Release.Name }} -n {{ .Release.Namespace }}
